● 基于对您文档和现有代码的分析，我来详细讨论这个任务管理问题：

  问题分析

  当前系统的痛点

  1. 同步阻塞问题：现在的 generate.js 调用 driver.generate() 会一直等待结果，对于慢速生成（如3D模型、视频），用户体验很差
  2. 驱动异构性：有些驱动（如 Meshy、LTX2）原生支持任务ID，而有些（如 Google）是同步的
  3. 状态管理缺失：没有统一的任务状态跟踪机制，用户无法查看进行中的任务

  两种方案对比

  方案A：直接在 live-gen.json 添加状态字段

  优点：
  - 实现简单，只需修改现有数据结构
  - 历史记录和任务状态在同一个地方
  - 不需要额外的数据迁移逻辑

  缺点：
  - 混淆了"进行中任务"和"已完成历史"的概念
  - 频繁更新大文件可能有性能问题
  - 难以区分失败任务和成功记录

  方案B：独立的任务队列系统

  优点：
  - 清晰的职责分离：任务队列负责执行，live-gen.json 只存储成功结果
  - 更好的并发控制和错误处理
  - 便于实现任务优先级、重试等高级功能
  - 可以轻松添加任务取消功能

  缺点：
  - 实现相对复杂
  - 需要维护两套数据结构

  推荐方案：混合模式

  class TaskManager {
    constructor() {
      this.tasksFile = path.resolve(__dirname, '../../data/tasks.json');
      this.ensureTaskFile();
    }

    async createTask(params) {
      const task = {
        id: randomUUID(),
        type: params.type,
        status: 'pending',
        modelId: params.modelId,
        prompt: params.prompt,
        params: params.params,
        createdAt: new Date().toISOString(),
        progress: 0
      };

      const tasks = await this.getTasks();
      tasks.push(task);
      await this.saveTasks(tasks);

      // 立即返回任务ID给前端
      return task;
    }

    async updateTaskStatus(taskId, status, progress, result, error) {
      const tasks = await this.getTasks();
      const task = tasks.find(t => t.id === taskId);
      if (task) {
        task.status = status;
        task.progress = progress || task.progress;
        task.result = result || task.result;
        task.error = error || task.error;
        task.updatedAt = new Date().toISOString();
        await this.saveTasks(tasks);
      }
      return task;
    }
  }

  const fs = require('fs/promises');
  const path = require('path');
  const { randomUUID } = require('crypto');

  class TaskManager {
    constructor() {
      this.tasksFile = path.resolve(__dirname, '../../data/tasks.json');
      this.tasks = [];
      this.init();
    }

    async init() {
      try {
        const data = await fs.readFile(this.tasksFile, 'utf-8');
        this.tasks = JSON.parse(data);
      } catch {
        this.tasks = [];
        await this.save();
      }
    }

    async save() {
      await fs.writeFile(this.tasksFile, JSON.stringify(this.tasks, null, 2));
    }

    async createTask(params) {
      const task = {
        id: randomUUID(),
        ...params,
        status: 'pending',
        progress: 0,
        createdAt: new Date().toISOString()
      };

      this.tasks.push(task);
      await this.save();

      // 模拟异步执行
      this.simulateExecution(task.id);

      return task;
    }

    async simulateExecution(taskId) {
      // 模拟任务执行
      for (let i = 1; i <= 10; i++) {
        await new Promise(r => setTimeout(r, 1000));
        await this.updateTask(taskId, {
          status: 'running',
          progress: i * 10
        });
      }

      await this.updateTask(taskId, {
        status: 'completed',
        progress: 100,
        result: { imagePath: `/imagedb/test/${taskId}.png` }
      });
    }

    async updateTask(taskId, updates) {
      const task = this.tasks.find(t => t.id === taskId);
      if (task) {
        Object.assign(task, updates);
        task.updatedAt = new Date().toISOString();
        await this.save();
      }
      return task;
    }

    getTask(taskId) {
      return this.tasks.find(t => t.id === taskId);
    }

    getTasks() {
      return this.tasks;
    }
  }

  module.exports = new TaskManager();

      const activeTasks = new Set();

      async function createTask(type, driver) {
        const task = await fetch('/api/tasks', {
          method: 'POST',

          body: JSON.stringify({

      async function pollTask(taskId) {
        const poll = async () => {
          const task = await fetch(`/api/tasks/${taskId}`).then(r => r.json());
          updateTaskCard(task);

          if (task.status === 'running' || task.status === 'pending') {
            setTimeout(poll, 1000);
          } else {
            activeTasks.delete(taskId);
          }
        };
        poll();

        const container = document.getElementById('tasks');
